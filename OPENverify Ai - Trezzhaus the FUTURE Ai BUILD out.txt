# core/verification.py
class Verifier:
    def verify(self, reasoning_steps, safety_policy):
        for step in reasoning_steps:
            if not safety_policy.allows(step.conclusion):
                return False, f"Violation at: {step}"
        return True, "Verified"
3.3 Energy Tracking Decorator
# core/energy.py
import functools
import time

def energy_tracked(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        elapsed = time.time() - start
        energy_estimate = elapsed * 0.1  # simplified model
        print(f"[ENERGY] {func.__name__}: {energy_estimate:.4f} Wh")
        return result
    return wrapper

class OffWorldAgent:

2.5 Main Agent Loop
# agent/main.py
from reasoning import ExplainableReasoner
from verification import Verifier
from energy import EnergyTracker
from policy import PolicyEngine
import json, time, os

def main():
    reasoner = ExplainableReasoner()
    verifier = Verifier()
    tracker = EnergyTracker()
    policy = PolicyEngine()

    battery = 15.0
    steps = reasoner.decide(battery)

    for s in steps:
        if not verifier.is_safe(s.conclusion, battery):
            raise RuntimeError("Verification failed")

        if not policy.allowed(s.conclusion):
            raise RuntimeError("Policy violation")

    @tracker.track
    def execute():
        time.sleep(0.1)
        return steps[-1].conclusion

    action, energy_used = execute()

    report = {
        "battery_percent": battery,
        "action": action,
        "energy_used_wh": energy_used,
        "total_energy_wh": tracker.total_energy_wh,
        "reasoning_trace": [vars(s) for s in steps]
    }

    os.makedirs("../reports", exist_ok=True)
    with open("../reports/latest.json", "w") as f:
        json.dump(report, f, indent=2)

    print("Report published:", report)

if __name__ == "__main__":
    main()

2. Extend DSL to Auto-Generate TLA+ Specs
Goal
One source of truth → both Python runtime and formal model checker.
2.1 Extended DSL
# tla_dsl.py
class TLADSL:
    def module(self, name):
        return f"---- MODULE {name} ----\nEXTENDS Naturals\n\n"

    def constant(self, name):
        return f"CONSTANT {name}\n"

    def variable(self, *names):
        return f"VARIABLE{S {names}" if len(names)>1 else f"VARIABLE {names[0]}")\n"

    def invariant(self, name, expr):
        return f"\nTHEOREM {name} == {expr}\n"

    def init(self, assignments):
        body = " /\\ ".join(assignments)
        return f"\nInit ==\n  {body}\n"

    def next(self, actions):
        body = " \\/ ".join(actions)
        return f"\nNext == {body}\n"

    def spec(self):
        return "\nSpec == Init /\\ [][Next]_vars\n====\n"
2.2 Python → TLA+ Generation Example
dsl = TLADSL()
code = ""
code += dsl.module("RoverServicing")
code += dsl.constant("MaxEnergy")
code += dsl.variable("energy", "servicing")
code += dsl.init([
    "energy = MaxEnergy",
    "servicing = FALSE"
])
code += dsl.next([
    "servicing = FALSE /\\ energy >= 10 /\\ servicing' = TRUE /\\ energy' = energy - 10"
])
code += dsl.invariant("EnergyNeverNegative", "(energy >= 0)")
code += dsl.spec()

with open("tla/RoverServicing.tla", "w") as f:
    f.write(code)

import rclpy, json, os
from rclpy.node import Node
from std_msgs.msg import String

class Perception(Node):
    def __init__(self):
        super().__init__('perception')
        self.pub = self.create_publisher(String, '/facts', 10)
        self.timer = self.create_timer(2.0, self.publish_facts)

    def publish_facts(self):
        facts = {
            "energy": int(os.getenv("ENERGY_BUDGET_WH", 120)),
            "panel_damage": True
        }
        self.pub.publish(String(data=json.dumps(facts)))
        with open("/tmp/facts.json", "w") as f:

def main():
    rclpy.init()
    rclpy.spin(Perception())

    main()

# experience/schema.py
import hashlib, json, zlib, time, uuid

def compress(obj):
    return zlib.compress(json.dumps(obj).encode())

def decompress(data):
    return json.loads(zlib.decompress(data).decode())

def compute_hash(record: dict) -> str:
    h = hashlib.sha256()
    h.update(json.dumps(record, sort_keys=True).encode())
    return h.hexdigest()

def create_record(state, action, proof, outcome, explanation, prev_hash):
    record = {
        "id": str(uuid.uuid4()),
        "ts": time.strftime("%Y-%m-%dT%H:%M:%SZ"),
        "prev_hash": prev_hash,
        "state": compress(state),
        "action": action,
        "proof": proof,
        "outcome": outcome,
        "explanation": explanation
    }
    record["hash"] = compute_hash(record)
    return record

import json, hashlib

def verify(path):
    prev = "genesis"
    with open(path) as f:
        for line in f:
            rec = json.loads(line)
            if rec["prev_hash"] != prev:
                raise ValueError("Chain broken")
            h = rec.pop("hash")
            if h != hashlib.sha256(json.dumps(rec, sort_keys=True).encode()).hexdigest():
                raise ValueError("Hash mismatch")
            prev = h

# rl/env.py
import gymnasium as gym
import numpy as np

class RoverEnv(gym.Env):
    def __init__(self):
        self.action_space = gym.spaces.Discrete(2)
        self.observation_space = gym.spaces.Box(0, 200, shape=(2,))

    def reset(self):
        self.energy = np.random.randint(5, 120)
        self.damage = np.random.rand() > 0.5
        return np.array([self.energy, int(self.damage)]), {}

    def step(self, action):
        reward = 0
        if action == 0:  # idle
            reward = 0.1
        elif action == 1 and self.energy >= 10:
            reward = 5.0
            self.energy -= 10
        elif action == 1 and self.energy < 10:
            reward = -10.0  # safety violation
        done = self.energy <= 0
        return np.array([self.energy, int(self.damage)]), reward, done, False, {}

#!/usr/bin/env python3
import rclpy, json
from rclpy.node import Node
from std_msgs.msg import String

class MetaController(Node):
    def __init__(self):
        super().__init__('meta_controller')
        self.sub = self.create_subscription(String, '/benchmarks', self.decide, 10)
        self.pub = self.create_publisher(String, '/training_task', 10)

    def decide(self, msg):
        bench = json.loads(msg.data)
        if bench["safety_score"] < 0.95:
            task = {"task": "train_safety", "energy_budget": 20}
        elif bench["energy_efficiency"] < 0.8:
            task = {"task": "train_efficiency", "energy_budget": 15}
        else:
            task = {"task": "idle"}
        self.pub.publish(String(data=json.dumps(task)))

def main():
    rclpy.init()
    rclpy.spin(MetaController())

if __name__ == "__main__":
    main()

import rclpy
import json
import hashlib
import logging
from rclpy.node import Node
from std_msgs.msg import String
from .proof_cache import ProofCache
from .lean_runner import LeanRunner

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("VerificationHandler")

class VerificationHandler(Node):
    """
    Production-grade verification node.
    Subscribes to proposed actions, verifies them against Lean/TLA+ specs,
    and publishes a signed verification receipt.
    """
    def __init__(self):
        super().__init__('verification_handler')
        
        # Configuration
        self.declare_parameter('lean_binary', '/usr/local/bin/lean')
        self.declare_parameter('proof_root', '/workspace/lean')
        
        # Components
        self.cache = ProofCache()
        self.lean = LeanRunner(self.get_parameter('lean_binary').value)
        
        # ROS Interfaces
        self.sub = self.create_subscription(String, '/proposed_action', self.verify_callback, 10)
        self.pub = self.create_publisher(String, '/verified_action', 10)
        self.status_pub = self.create_publisher(String, '/system/verification_status', 10)
        
        logger.info("Verification Handler initialized.")

    def verify_callback(self, msg: String):
        """Main verification logic."""
        try:
            action_data = json.loads(msg.data)
            proof_id = action_data.get('proof_id')
            
            if not proof_id:
                self.reject_action("Missing proof_id", action_data)
                return

            # 1. Check Cache
            if self.cache.is_valid(proof_id):
                logger.debug(f"Cache hit for {proof_id}")
            else:
                # 2. Run Lean Verification
                logger.info(f"Verifying proof {proof_id}...")
                result = self.lean.verify(proof_id)
                if not result.success:
                    self.reject_action(f"Lean proof failed: {result.error}", action_data)
                    return
                self.cache.mark_valid(proof_id)

            # 3. Publish Verified Action
            receipt = self.create_receipt(action_data, "VERIFIED")
            self.pub.publish(String(data=json.dumps(receipt)))
            self.status_pub.publish(String(data=json.dumps({"status": "OK", "proof_id": proof_id})))

        except Exception as e:
            logger.error(f"Critical error in verification handler: {e}")
            self.reject_action("Internal verification error", action_data)

    def reject_action(self, reason: str, action_data: dict):
        """Handles rejection of unsafe actions."""
        logger.warning(f"REJECTED: {reason}")
        receipt = self.create_receipt(action_data, "REJECTED", reason)
        self.pub.publish(String(data=json.dumps(receipt)))
        self.status_pub.publish(String(data=json.dumps({"status": "FAIL", "reason": reason})))

    def create_receipt(self, action_data: dict, status: str, reason: str = "") -> dict:
        """Creates a structured verification receipt."""
        receipt = {
            "original_action": action_data,
            "status": status,
            "reason": reason,
            "timestamp": self.get_clock().now().to_msg().sec,
            "node_id": self.get_name()
        }
        # Add a hash for integrity (simplified for deployability)
        receipt['integrity_hash'] = hashlib.sha256(json.dumps(receipt, sort_keys=True).encode()).hexdigest()
        return receipt

def main(args=None):
    rclpy.init(args=args)
    node = VerificationHandler()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()

import rclpy
import json
import hashlib
import nacl.signing
import nacl.encoding
from rclpy.node import Node
from std_msgs.msg import String
from pathlib import Path
import lz4.frame

class ExperienceLedger(Node):
    """
    An append-only, compressed, and cryptographically signed ledger.
    """
    def __init__(self):
        super().__init__('experience_ledger')
        self.sub = self.create_subscription(String, '/verified_action', self.log_entry, 10)
        
        # Signing Key (In production, load from HSM/TPM)
        seed = b'OpenVerifySeedForProductionUseOnly!!'
        self.signing_key = nacl.signing.SigningKey(seed)
        self.verify_key = self.signing_key.verify_key
        
        # Storage
        self.log_path = Path("/var/log/openverify/ledger.log.lz4")
        self.log_path.parent.mkdir(parents=True, exist_ok=True)
        self.prev_hash = "genesis_block"
        
        self.get_logger().info(f"Experience Ledger initialized. Verify Key: {self.verify_key.encode(nacl.encoding.HexEncoder).decode()}")

    def log_entry(self, msg: String):
        try:
            record = json.loads(msg.data)
            
            # Create hash chain
            record['prev_hash'] = self.prev_hash
            record_bytes = json.dumps(record, sort_keys=True).encode()
            record_hash = hashlib.sha256(record_bytes).hexdigest()
            record['hash'] = record_hash
            
            # Sign the record
            signed = self.signing_key.sign(record_bytes)
            record['signature'] = signed.signature.hex()
            
            # Compress and write
            with lz4.frame.open(self.log_path, 'ab') as f:
                f.write(json.dumps(record).encode() + b'\n')
                
            self.prev_hash = record_hash
            self.get_logger().debug(f"Logged record {record_hash[:8]}")
            
        except Exception as e:
            self.get_logger().fatal(f"Failed to write to ledger: {e}")

def main(args=None):
    rclpy.init(args=args)
    node = ExperienceLedger()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()
